            function shuffleArray(arr) {
                for (let i = arr.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [arr[i], arr[j]] = [arr[j], arr[i]];
                }
                return arr;
            }

            // ---------- 生成拼图 ----------
            function generatePuzzle(numWords) {
                const gridSize = state.gridSize;
                const grid = Array.from({ length: gridSize }, () => Array(gridSize).fill(''));
                const positions = [];
                const shuffled = shuffleArray([...WORD_BANK]);
                const unique = [];
                const seen = new Set();
                for (let w of shuffled) {
                    if (!seen.has(w) && w.length >= 3 && w.length <= 8) {
                        seen.add(w);
                        unique.push(w);
                    }
                }
                const words = unique.slice(0, numWords);
                const dirs = [
                    [0, 1],
                    [1, 0],
                    [1, 1],
                    [1, -1]
                ];
                for (let w of words) {
                    let placed = false;
                    let attempts = 0;
                    while (!placed && attempts < 300) {
                        attempts++;
                        const row = Math.floor(Math.random() * gridSize);
                        const col = Math.floor(Math.random() * gridSize);
                        const dir = dirs[Math.floor(Math.random() * dirs.length)];
                        const dr = dir[0],
                            dc = dir[1];
                        let fit = true;
                        for (let i = 0; i < w.length; i++) {
                            const r = row + i * dr,
                                c = col + i * dc;
                            if (r < 0 || r >= gridSize || c < 0 || c >= gridSize) { fit = false; break; }
                            if (grid[r][c] !== '' && grid[r][c] !== w[i]) { fit = false; break; }
                        }
                        if (fit) {
                            const cells = [];
                            for (let i = 0; i < w.length; i++) {
                                const r = row + i * dr,
                                    c = col + i * dc;
                                grid[r][c] = w[i];
                                cells.push([r, c]);
                            }
                            positions.push({ word: w, row, col, dir: dir, cells });
                            placed = true;
                        }
                    }
                    if (!placed) console.warn('无法放置单词:', w);
                }
                for (let r = 0; r < gridSize; r++) {
                    for (let c = 0; c < gridSize; c++) {
                        if (grid[r][c] === '') {
                            grid[r][c] = String.fromCharCode(65 + Math.floor(Math.random() * 26));
                        }
                    }
                }
                return { grid, positions, words: positions.map(p => p.word) };
            }

            function renderGrid() {

                let html = '';
                for (let r = 0; r < size; r++) {
                    html += '<tr>';
                    for (let c = 0; c < size; c++) {

            // ---------- 触摸滑动（全新实现：基于起点方向，填充直线，支持斜线） ----------
            let touchActive = false;
            let startRow = -1,
                startCol = -1;

            function setupGridTouchEvents() {
                const tds = gridTable.querySelectorAll('td');
                tds.forEach(td => {
                    td.removeEventListener('touchstart', onTouchStart);
                    td.removeEventListener('touchmove', onTouchMove);
                    td.removeEventListener('touchend', onTouchEnd);
                    td.removeEventListener('touchcancel', onTouchEnd);
                });
                tds.forEach(td => {
                    td.addEventListener('touchstart', onTouchStart, { passive: true });
                    td.addEventListener('touchmove', onTouchMove, { passive: true });
                    td.addEventListener('touchend', onTouchEnd, { passive: true });
                    td.addEventListener('touchcancel', onTouchEnd, { passive: true });
                });
            }

            function onTouchStart(e) {
                if (state.gameOver || state.isWaiting) return;
                const td = e.currentTarget;
                const row = parseInt(td.dataset.row);
                const col = parseInt(td.dataset.col);
                if (isCellFound(row, col)) {
                    touchActive = false;
                    return;
                }
                touchActive = true;
                startRow = row;
                startCol = col;
                state.selectedCells = [
                    [row, col]
                ];
                renderGrid();
            }

            function onTouchMove(e) {
                if (!touchActive || state.gameOver || state.isWaiting) return;
                const touch = e.touches[0];
                const el = document.elementFromPoint(touch.clientX, touch.clientY);
                if (!el) return;
                const td = el.closest('td');
                if (!td) return;
                const row = parseInt(td.dataset.row);
                const col = parseInt(td.dataset.col);
                // 不能选中已找到的格子
                if (isCellFound(row, col)) return;

                // 计算从起点到当前点的方向
                const dr = Math.sign(row - startRow);
                const dc = Math.sign(col - startCol);
                // 方向必须为 (0,±1), (±1,0), (±1,±1) 之一，且不能同时为0
                if (dr === 0 && dc === 0) return;
                // 检查当前点是否在从起点沿该方向的直线上（即与起点的行差列差比例一致）
                // 由于我们只允许水平、垂直、对角线，所以只要 (row-startRow) 和 (col-startCol) 的绝对值成比例
                const absDr = Math.abs(row - startRow);
                const absDc = Math.abs(col - startCol);
                // 必须满足：要么行差为0，要么列差为0，要么行差等于列差（对角线）
                if (!(absDr === 0 || absDc === 0 || absDr === absDc)) {
                    return; // 不在直线方向上
                }

                // 沿直线填充从起点到当前点的所有格子（不包括起点，起点已在selectedCells中）
                const newCells = [];
                const steps = Math.max(absDr, absDc);
                // 从第1步到第steps步
                for (let i = 1; i <= steps; i++) {
                    const r = startRow + i * dr;
                    const c = startCol + i * dc;
                    if (r < 0 || r >= state.gridSize || c < 0 || c >= state.gridSize) break;
                    if (isCellFound(r, c)) break; // 遇到已找到的格子停止
                    newCells.push([r, c]);
                }
                // 更新selectedCells为起点+中间点
                state.selectedCells = [
                    [startRow, startCol]
                ];
                for (let cell of newCells) {
                    state.selectedCells.push(cell);
                }
                renderGrid();
            }

            function onTouchEnd(e) {
                if (touchActive && state.selectedCells.length >= 3) {
                    checkSelectedWord();
                }
                touchActive = false;
                startRow = -1;
                startCol = -1;
            }

            // ---------- 检查单词（验证共线连续） ----------
            function checkSelectedWord() {
                if (state.selectedCells.length < 3) {
                    state.selectedCells = [];
                    renderGrid();
                    return;
                }
                const sorted = state.selectedCells;
                // 验证所有点是否在一条直线上且连续（步长1）
                let isValid = true;
                if (sorted.length >= 2) {
                    const dr = sorted[1][0] - sorted[0][0];
                    const dc = sorted[1][1] - sorted[0][1];
                    if (dr === 0 && dc === 0) isValid = false;
                    else {
                        for (let i = 1; i < sorted.length; i++) {
                            const ddr = sorted[i][0] - sorted[i - 1][0];
                            const ddc = sorted[i][1] - sorted[i - 1][1];
                            if (ddr !== dr || ddc !== dc) { isValid = false; break; }
                            if (Math.abs(ddr) > 1 || Math.abs(ddc) > 1) { isValid = false; break; }
                        }
                    }
                }
                if (!isValid) {
                    state.selectedCells = [];
                    renderGrid();
                    showToast('请沿同一方向连续滑动选择');
                    return;
                }

                let word = '';
                for (let [r, c] of sorted) {
                    word += state.grid[r][c];
                }
                word = word.toLowerCase();

                const wordObj = state.wordPositions.find(p => p.word === word && !state.found.includes(word));
                if (wordObj) {
                    state.found.push(word);
                    state.score += 10;
                    state.selectedCells = [];
                    renderGrid();
                    renderWordList();
                    updateUI();
                    showToast(`✅ 找到 "${word}"！+10 分`);
                    aiReact('found', word);
                    saveProgress();
                    if (state.found.length === state.words.length) {
                        setTimeout(() => finishGame(), 600);
                    }
                } else {
                    state.selectedCells = [];
                    renderGrid();
                    showToast('不是目标单词，再试试');
                }
            }

            function isCellFound(row, col) {
                for (let p of state.wordPositions) {
                    if (state.found.includes(p.word)) {
                        for (let [r, c] of p.cells) {
                            if (r === row && c === col) return true;
                        }
                    }
                }
                return false;
            }

            async function callAIForResponse(prompt, context) {
                if (!isApiAvailable()) {
                    return getFallbackResponse(context);
                }
                try {
                    let messages = [];
                    if (state.rolePackage && state.rolePackage.messages) {
                        messages = [...state.rolePackage.messages];
                    } else {
                        messages = [{
                            role: 'system',
                            content: `你是一个温暖、风趣、善于鼓励的搭档，名叫${state.charName}。你的回答简短、有创意，贴合你的人设。用中文回复。绝对不要使用任何动作描述（如*微笑*），不要用星号或括号，只输出纯自然语言对话。`
                        }];
                    }
                    messages.push({
                        role: 'system',
                        content: `重要：你的回复必须只包含自然语言，不要有任何动作描述（如*微笑*）、不要用星号、括号或任何格式标记。只输出对话内容。游戏是英文单词搜索。`
                    });
                    const progress = `${state.found.length}/${state.words.length} 个单词已找到`;
                    messages.push({
                        role: 'system',
                        content: `玩家：${state.playerName}。当前游戏进度：${progress}。${context}`
                    });
                    messages.push({ role: 'user', content: prompt });

                    const result = await API.callLLM({
                        characterId: state.selectedCharId,
                        messages: messages,
                    });
                    let reply = result.content || getFallbackResponse(context);
                    reply = reply.replace(/\*[^*]*\*/g, '').replace(/\([^)]*\)/g, '').trim();
                    return reply;
                } catch (e) {
                    console.warn(e);
                    return getFallbackResponse(context);
                }
            }

            function getFallbackResponse(context) {
                if (context.includes('greeting')) {
                    return `嗨，${state.playerName}！来挑战一下眼力吧，找到所有隐藏的单词！`;

                    const msgs = ['🎉 太棒了！', '🌟 好厉害！', '👏 真不错！', '💪 你太强了！'];
                    return msgs[Math.floor(Math.random() * msgs.length)];
                } else if (context.includes('重置')) {
                    return '😊 没关系，慢慢来。';
                } else if (context.includes('所有单词')) {
                    return '🎊 你做到了！太了不起了！';
                }
                return '😊 继续加油！';
            }

            // ---------- 提示 ----------
            async function giveHint() {
                if (state.gameOver || state.isWaiting) return;
                if (state.hintRemaining <= 0) { showToast('提示次数已用完'); return; }
                state.hintRemaining -= 1;
                const unfound = state.words.filter(w => !state.found.includes(w));
                if (unfound.length === 0) { showToast('所有单词都已找到！'); return; }
                const word = unfound[Math.floor(Math.random() * unfound.length)];
                const pos = state.wordPositions.find(p => p.word === word);
                if (!pos) { showToast('出错了，请重试'); return; }
                const [r, c] = pos.cells[0];
                const hintText = `有一个单词以 "${word[0].toUpperCase()}" 开头，共 ${word.length} 个字母，从第 ${r+1} 行第 ${c+1} 列开始。`;
                state.selectedCells = [];
                renderGrid();
                const td = gridTable.querySelector(`td[data-row="${r}"][data-col="${c}"]`);
                if (td) {
                    td.classList.add('highlight');
                    setTimeout(() => td.classList.remove('highlight'), 3000);
                }
                updateUI();
                saveProgress();

                state.isWaiting = true;
                setAiMessage(state.charName, '🤔 让我想想怎么提示你…', true);
                const prompt =
                    `你刚刚给${state.playerName}一个提示：${hintText}。请用你的口吻，把这句话说得自然、贴合你的人设，像是你在给对方线索。不要添加额外无关内容，只说这一句提示。`;
                const reply = await callAIForResponse(prompt, '提示');
                setAiMessage(state.charName, reply);
                state.isWaiting = false;
                updateUI();
                showToast(`💡 提示：${hintText}`);
            }

                document.querySelector('.result-screen .big-icon').textContent = icon;

                const context = '找到了所有单词';
                const prompt =
                    `${state.playerName} 找到了全部 ${total} 个单词，得分 ${score}。请根据你的人设，说一段温暖的总结和告别。只用自然语言。`;
                const reply = await callAIForResponse(prompt, context);
                resultAiMsg.textContent = reply;

                await recordGameResult(score);
                await safeApiCall(() => API.saveGame({ wordsearch_data: null }), null);
            }

            async function recordGameResult(score) {
                if (!isApiAvailable()) return;
                try {
                    const pName = state.playerName || '玩家';
                    const cName = state.charName || '搭档';
                    const summary =
                        `${pName} 和 ${cName} 一起玩了英文单词搜索游戏，${pName} 找到了所有单词，获得 ${score} 分。`;
                    await API.recordGameEvent({
                        characterIds: [state.selectedCharId],
                        summary: summary,
                    });
                } catch (e) { console.warn(e); }
            }

            // ---------- 存档 ----------
            async function saveProgress() {
                if (!isApiAvailable()) return;
                try {
                    await API.saveGame({
                        wordsearch_data: {
                            grid: state.grid,
                            wordPositions: state.wordPositions,
                            words: state.words,
                            found: state.found,
                            score: state.score,
                            hintRemaining: state.hintRemaining,
                            gameOver: state.gameOver,
                        }
                    });
                } catch (e) { console.warn(e); }
            }

            // ---------- 重新开始 / 退出 ----------
            async function replayGame() {
                resultScreen.classList.remove('active');
                gamePlayScreen.classList.add('active');
                resetGameState();
                renderGrid();
                renderWordList();
                updateUI();
                aiGreeting();
                await safeApiCall(() => API.saveGame({ wordsearch_data: null }), null);
            }

            async function closeGame() {
                if (isApiAvailable()) {
                    try { await API.closeGame(); } catch (e) { console.warn(e); }
                } else {
                    showToast('👋 再见！');
                    setTimeout(() => {
                        document.body.innerHTML =
                            '<div style="display:flex;align-items:center;justify-content:center;height:100%;color:rgba(255,255,255,0.3);font-size:16px;background:#0d0b1a;">游戏已退出</div>';
                    }, 500);
                }
            }

            // ---------- 初始化 ----------
            async function init() {
                if (isApiAvailable()) {
                    try {
                        await API.setTitleBar({
                            material: 'glass',
                            background: 'rgba(13, 11, 26, 0.72)',
                            color: '#f0ecf5',
                            borderColor: 'rgba(255,255,255,0.06)',
                            buttonBackground: 'rgba(255,255,255,0.06)',
                            buttonColor: '#f0ecf5',
                            buttonBorderColor: 'rgba(255,255,255,0.08)',
                            buttonRadius: '999px',
                            buttonShadow: '0 4px 16px rgba(0,0,0,0.3)',
                            iconOpacity: 0.9,
                        });
                    } catch (e) { /* ignore */ }
                }

                await loadCharacters();

                startBtn.addEventListener('click', startGame);
                hintBtn.addEventListener('click', giveHint);
                resetBtn.addEventListener('click', resetPuzzle);
                replayBtn.addEventListener('click', replayGame);
                closeBtn.addEventListener('click', closeGame);

                setTimeout(() => {
                    if (charSelectScreen.style.display !== 'none' && state.characters.length > 0 && !state
                        .selectedCharId) {
                        selectCharacter(state.characters[0].id);
                    }
                }, 400);

                console.log('[Word Search] 滑动优化：基于起点方向，稳定支持斜线选择，减少误触。');
            }

            if (document.readyState === 'complete' || document.readyState === 'interactive') {
                init();
            } else {
                document.addEventListener('DOMContentLoaded', init);
            }

        })();
    </script>
